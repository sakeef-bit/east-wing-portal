Please authenticate."); } const glContext = appGLOptions.length > 0 ? appGLOptions.join(", ") : "Standard GAAP Accounts (Payroll Expense, Rent Expense, Utilities, Office Expense, Bank Fees, Consulting Income)"; const pastHistoryContext = historyData.length > 0 ? historyData.slice(0, 30).map(h => `${h.vendor || h.desc} -> ${h.cat}`).join("\n") : "No historical patterns recorded yet."; const behavior = type === "CC" ? "Spend is negative, Payments positive." : "Debits are negative, Credits positive."; const prompt = `ACT AS: Senior Bookkeeper & Forensic Auditor. Extract all statement transactions. CHART OF ACCOUNTS (MUST SELECT FROM THIS LIST): [${glContext}] HISTORICAL CATEGORIZATION PATTERNS: ${pastHistoryContext} STRICT EXTRACTION RULES: 1. 'vendor': Extract the clean entity or merchant name from the statement line (e.g., 'Con Edison', 'ACME Corp', 'Local Supermarket', 'Stripe', 'H. Sharma', 'Adobe', 'Uber'). Only set vendor to "" if it is a pure generic internal fee or generic direct deposit with no entity name. 2. 'desc': Store the exact raw statement line text. 3. 'category': Categorize EVERY transaction using the provided CHART OF ACCOUNTS list above. Match rent to Rent Expense, payroll to Payroll Expense, utilities to Utilities, catering/supermarket to Office/Meals Expense, wire in to Revenue/Income, and fees to Bank Fees. ${behavior} Return ONLY JSON Object: { "starting_balance": 0.0, "ending_balance": 0.0, "transactions": [ { "date": "YYYY-MM-DD", "desc": "raw statement line", "vendor": "Clean Entity or Merchant Name", "amount": 0.00, "category": "Selected GL Account Name" } ] }`; const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-flash:generateContent?key=${apiKey}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ "contents": [{ "parts": [{ "text": prompt }, { "inline_data": { "mime_type": "application/pdf", "data": pdfData }}] }], "generationConfig": { "response_mime_type": "application/json" } }) }); const jsonRes = await res.json(); if (jsonRes.error) throw new Error("Gemini API Error: " + jsonRes.error.message); if (!jsonRes.candidates?.[0]?.content?.parts?.[0]?.text) throw new Error("Invalid response from Gemini."); const rawText = jsonRes.candidates[0].content.parts[0].text; const extracted = JSON.parse(rawText.replace(/```json|```/g, '').trim()); stmtStart = extracted.starting_balance || 0; stmtEnd = extracted.ending_balance || 0; await db.from('client_metadata').update({ stmt_start: stmtStart, stmt_end: stmtEnd }).eq('client_name', currentClient); const rows = (extracted.transactions || []).map(t => { let assignedCategory = t.category || "Uncategorized"; activeRules.forEach(rule => { const targetText = ((t.vendor || "") + " " + (t.desc || "")).toLowerCase(); if(targetText.includes(rule.keyword.toLowerCase())) { assignedCategory = rule.target_account; } }); return { client_name: currentClient, source: type === 'PDF' ? 'BANK' : 'CREDIT CARD', txn_date: t.date, vendor: t.vendor || "", description: t.desc || "Statement Line", amount: t.amount, category: assignedCategory, status: "PENDING", bank_category: "Uncategorized", class_name: "General", fingerprint: btoa(t.date + (t.desc || "") + t.amount) }; }); if (rows.length === 0) { document.getElementById('st').style.display = 'none'; return alert("No valid transactions found."); } await db.from('review_staging').insert(rows); await loadDash(); } catch(err) { alert("Processing Error: " + err.message); document.getElementById('st').style.display = 'none'; } }; reader.readAsDataURL(input.files[0]); }
